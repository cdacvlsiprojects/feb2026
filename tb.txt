`timescale 1ns/1ps

// ------------------------------------------------------------
// Simple AXI-Lite memory slave (SV)
// - 32-bit data, 64KB default
// - 1 outstanding read + 1 outstanding write response
// - fixed response latency (RESP_LATENCY cycles)
// ------------------------------------------------------------
module axil_mem_slave_sv #(
  parameter int AW = 32,
  parameter int DW = 32,
  parameter int MEM_BYTES = 64*1024,
  parameter int RESP_LATENCY = 2
)(
  input  logic            ACLK,
  input  logic            ARESETN,

  input  logic            S_AWVALID,
  output logic            S_AWREADY,
  input  logic [AW-1:0]   S_AWADDR,
  input  logic [2:0]      S_AWPROT,

  input  logic            S_WVALID,
  output logic            S_WREADY,
  input  logic [DW-1:0]   S_WDATA,
  input  logic [DW/8-1:0] S_WSTRB,

  output logic            S_BVALID,
  input  logic            S_BREADY,
  output logic [1:0]      S_BRESP,

  input  logic            S_ARVALID,
  output logic            S_ARREADY,
  input  logic [AW-1:0]   S_ARADDR,
  input  logic [2:0]      S_ARPROT,

  output logic            S_RVALID,
  input  logic            S_RREADY,
  output logic [DW-1:0]   S_RDATA,
  output logic [1:0]      S_RRESP
);
  localparam int WORDS = MEM_BYTES/4;
  logic [31:0] mem [0:WORDS-1];

  logic        have_aw, have_w, have_ar;
  logic [AW-1:0] awaddr_q, araddr_q;
  logic [DW-1:0] wdata_q;
  logic [DW/8-1:0] wstrb_q;
  int w_cnt, r_cnt;

  // 64KB region => use [15:2] as word index for default mapping
  wire [31:0] aw_word_index = awaddr_q[15:2];
  wire [31:0] ar_word_index = araddr_q[15:2];

  assign S_BRESP = 2'b00; // OKAY
  assign S_RRESP = 2'b00; // OKAY

  always_comb begin
    S_AWREADY = ARESETN && !have_aw;
    S_WREADY  = ARESETN && !have_w;
    // keep one outstanding read at a time
    S_ARREADY = ARESETN && !have_ar && !S_RVALID;
  end

  always_ff @(posedge ACLK) begin
    if (!ARESETN) begin
      have_aw <= 1'b0; have_w <= 1'b0; have_ar <= 1'b0;
      awaddr_q <= '0; araddr_q <= '0;
      wdata_q <= '0; wstrb_q <= '0;
      S_BVALID <= 1'b0;
      S_RVALID <= 1'b0;
      S_RDATA  <= '0;
      w_cnt <= 0; r_cnt <= 0;

      for (int i=0; i<WORDS; i++) mem[i] <= 32'h0;
    end else begin
      // Latch AW
      if (S_AWVALID && S_AWREADY) begin
        have_aw  <= 1'b1;
        awaddr_q <= S_AWADDR;
      end

      // Latch W
      if (S_WVALID && S_WREADY) begin
        have_w  <= 1'b1;
        wdata_q <= S_WDATA;
        wstrb_q <= S_WSTRB;
      end

      // When both captured, perform write then start B latency counter
      if (have_aw && have_w && !S_BVALID && (w_cnt==0)) begin
        if (aw_word_index < WORDS) begin
          logic [31:0] cur;
          cur = mem[aw_word_index];
          if (wstrb_q[0]) cur[ 7: 0] = wdata_q[ 7: 0];
          if (wstrb_q[1]) cur[15: 8] = wdata_q[15: 8];
          if (wstrb_q[2]) cur[23:16] = wdata_q[23:16];
          if (wstrb_q[3]) cur[31:24] = wdata_q[31:24];
          mem[aw_word_index] <= cur;
        end
        w_cnt <= RESP_LATENCY;
      end

      if (w_cnt > 0) begin
        w_cnt <= w_cnt - 1;
        if (w_cnt == 1) S_BVALID <= 1'b1;
      end

      if (S_BVALID && S_BREADY) begin
        S_BVALID <= 1'b0;
        have_aw  <= 1'b0;
        have_w   <= 1'b0;
      end

      // Latch AR
      if (S_ARVALID && S_ARREADY) begin
        have_ar  <= 1'b1;
        araddr_q <= S_ARADDR;
        r_cnt    <= RESP_LATENCY;
      end

      if (have_ar && (r_cnt > 0)) begin
        r_cnt <= r_cnt - 1;
        if (r_cnt == 1) begin
          if (ar_word_index < WORDS) S_RDATA <= mem[ar_word_index];
          else                       S_RDATA <= 32'hDEAD_BEEF;
          S_RVALID <= 1'b1;
        end
      end

      if (S_RVALID && S_RREADY) begin
        S_RVALID <= 1'b0;
        have_ar  <= 1'b0;
      end
    end
  end
endmodule

  // ---------------- AXI-Lite master tasks (compliant & robust) ----------------

  

  // Cleanup/idle: prevents leftover valids/payloads from causing "ghost" transfers
  task automatic master_idle(input int m);
    begin
      S_AXI_AWVALID[m] = 1'b0;
      S_AXI_WVALID[m]  = 1'b0;
      S_AXI_ARVALID[m] = 1'b0;
      S_AXI_BREADY[m]  = 1'b0;

      S_AXI_AWPROT[m*3  +: 3 ] = '0;
      S_AXI_ARPROT[m*3  +: 3 ] = '0;

      // Give DUT 1 cycle to settle before next transaction
      @(posedge clk);
    end
  endtask

task automatic axil_write(
  input int m,
  input logic [AW-1:0] addr,
  input logic [DW-1:0] data,
  input logic [DW/8-1:0] strb
);
  int tmo;
  begin
    // --------------------
    // Drive address
    // --------------------
    S_AXI_AWADDR[m*AW +: AW] = addr;
    S_AXI_AWPROT[m*3  +: 3 ] = 3'b000;
    S_AXI_AWVALID[m] = 1'b1;

    // Let DUT sample AWVALID
    @(posedge clk);

    // Wait for AW handshake
    tmo = 0;
    while (!(S_AXI_AWVALID[m] && S_AXI_AWREADY[m])) begin
      @(posedge clk);
      if (!rstn) continue;
      tmo++;
      if (tmo > TIMEOUT_CYCLES) begin
        $error("[%0t] TIMEOUT in axil_write: M%0d addr=%h (AW not accepted)",
               $time, m, addr);
        disable axil_write;
      end
    end

   /////////// $display("[%0t] M%0d AW accepted addr=%h", $time, m, addr);
    S_AXI_AWVALID[m] = 1'b0;

    // --------------------
    // Drive write data
    // --------------------
    S_AXI_WDATA[m*DW +: DW] = data;
    S_AXI_WSTRB[m*(DW/8) +: (DW/8)] = strb;
    S_AXI_WVALID[m] = 1'b1;

    @(posedge clk);

    // Wait for W handshake
    tmo = 0;
    while (!(S_AXI_WVALID[m] && S_AXI_WREADY[m])) begin
      @(posedge clk);
      if (!rstn) continue;
      tmo++;
      if (tmo > TIMEOUT_CYCLES) begin
        $error("[%0t] TIMEOUT in axil_write: M%0d addr=%h (W not accepted)",
               $time, m, addr);
        disable axil_write;
      end
    end

    ///////$display("[%0t] M%0d W accepted data=%h", $time, m, data);
    S_AXI_WVALID[m] = 1'b0;

    // --------------------
    // Wait for response
    // --------------------
    S_AXI_BREADY[m] = 1'b1;

    tmo = 0;
    while (!S_AXI_BVALID[m]) begin
      @(posedge clk);
      if (!rstn) continue;
      tmo++;
      if (tmo > TIMEOUT_CYCLES) begin
        $error("[%0t] TIMEOUT in axil_write: M%0d addr=%h (BVALID never asserted)",
               $time, m, addr);
        disable axil_write;
      end
    end

    ///////$display("[%0t] M%0d WRITE addr=%h data=%h bresp=%b",$time, m, addr, data, S_AXI_BRESP[m*2 +: 2]);

    // Complete B handshake
    while (!(S_AXI_BVALID[m] && S_AXI_BREADY[m])) @(posedge clk);
    S_AXI_BREADY[m] = 1'b0;

    // --------------------
    // Clean up master signals (important!)
    // --------------------
    master_idle(m);
  end
endtask


task automatic axil_read(
  input int m,
  input logic [AW-1:0] addr,
  output logic [DW-1:0] data_out,
  output logic [1:0] resp_out
);
  int tmo;
  begin
    data_out = '0;
    resp_out = '0;

    // Drive AR and keep stable until handshake ON A CLOCK EDGE
    S_AXI_ARADDR[m*AW +: AW] = addr;
    S_AXI_ARPROT[m*3  +: 3 ] = 3'b000;
    S_AXI_ARVALID[m] = 1'b1;
    S_AXI_RREADY[m]  = 1'b1;   // OK to assert early

    // IMPORTANT: wait at least one posedge so DUT can sample ARVALID
    @(posedge clk);

    // Wait for AR handshake on a posedge
    tmo = 0;
    while (!(S_AXI_ARVALID[m] && S_AXI_ARREADY[m])) begin
      @(posedge clk);
      if (!rstn) continue;
      tmo++;
      if (tmo > TIMEOUT_CYCLES) begin
        $error("[%0t] TIMEOUT axil_read: M%0d addr=%h (AR not accepted) ARREADY=%b",
               $time, m, addr, S_AXI_ARREADY[m]);
        disable axil_read;
      end
    end

    ///////$display("[%0t] M%0d AR accepted addr=%h", $time, m, addr);
    S_AXI_ARVALID[m] = 1'b0;

    // Wait for RVALID (posedge-based)
    tmo = 0;
    while (!S_AXI_RVALID[m]) begin
      @(posedge clk);
      if (!rstn) continue;
      tmo++;
      if (tmo > TIMEOUT_CYCLES) begin
        $error("[%0t] TIMEOUT axil_read: M%0d addr=%h (RVALID never asserted)", $time, m, addr);
        disable axil_read;
      end
    end

    // Sample data/resp when RVALID is high
    data_out = S_AXI_RDATA[m*DW +: DW];
    resp_out = S_AXI_RRESP[m*2 +: 2];
    ////////$display("[%0t] M%0d READ  addr=%h data=%h rresp=%b",$time, m, addr, data_out, resp_out);

    // Complete handshake then drop RREADY
    while (!(S_AXI_RVALID[m] && S_AXI_RREADY[m])) @(posedge clk);
    S_AXI_RREADY[m] = 1'b0;
  end
endtask


  // ---------------- Scenarios ----------------
  logic [31:0] rd;
  logic [1:0]  rresp;
  logic [31:0] rd0, rd1;
  logic [1:0]  rp0, rp1;
  initial begin
    wait (rstn);
    repeat (5) @(posedge clk);

    $display("---- SCENARIO 1: Parallel writes (M0->S0) and (M1->S1) ----");
    fork
      axil_write(0, 32'h0000_0010, 32'hAAAA_0001, 4'hF);
      axil_write(1, 32'h0001_0020, 32'hBBBB_0001, 4'hF);
    join
    repeat (1) @(posedge clk);

    $display("---- SCENARIO 2: Contention (both masters write S0) ----");
    fork
      axil_write(0, 32'h0000_0030, 32'hAAAA_0030, 4'hF);
      axil_write(1, 32'h0000_0034, 32'hBBBB_0034, 4'hF);
    join
    repeat (1) @(posedge clk);

    $display("---- SCENARIO 3: Read-after-write correctness ----");
    axil_write(0, 32'h0001_0040, 32'h1234_5678, 4'hF);
    axil_read (0, 32'h0001_0040, rd, rresp);
    repeat (1) @(posedge clk);

    $display("---- SCENARIO 4: Illegal address (expect DECERR from xbar) ----");
    axil_write(1, 32'h0002_0000, 32'hDEAD_BEEF, 4'hF);
    axil_read (1, 32'h0002_0000, rd, rresp);
    repeat (1) @(posedge clk);

    $display("---- SCENARIO 5: Parallel reads (M0->S0) and (M1->S1) ----");

     // Preload both slaves
     axil_write(0, 32'h0000_0100, 32'h1111_0100, 4'hF); // S0
     axil_write(1, 32'h0001_0100, 32'h2222_0100, 4'hF); // S1

     repeat (1) @(posedge clk);

     fork
     axil_read(0, 32'h0000_0100, rd0, rp0);
     axil_read(1, 32'h0001_0100, rd1, rp1);
     join

    $display("S5 Results: M0 data=%h resp=%b | M1 data=%h resp=%b", rd0, rp0, rd1, rp1);

    $display("---- SCENARIO 6: Read contention (M0->S0, M1->S0) ----");

     // Preload S0
     axil_write(0, 32'h0000_0200, 32'hAAAA_0200, 4'hF);
     axil_write(0, 32'h0000_0204, 32'hBBBB_0204, 4'hF);

     repeat (1) @(posedge clk);

     fork
       axil_read(0, 32'h0000_0200, rd0, rp0);
       axil_read(1, 32'h0000_0204, rd1, rp1);
     join

    $display("S6 Results: M0 data=%h resp=%b | M1 data=%h resp=%b", rd0, rp0, rd1, rp1);
 
    $display("---- SCENARIO 7: Mixed RW contention on S0 (M0 write, M1 read) ----");

    // Preload read location
    axil_write(0, 32'h0000_0300, 32'h1234_0300, 4'hF);
    repeat (1) @(posedge clk);

   fork
     axil_write(0, 32'h0000_0310, 32'hDEAD_0310, 4'hF);  // M0 write S0
     axil_read (1, 32'h0000_0300, rd1, rp1);             // M1 read S0
   join

   $display("S7 Results: M1 read data=%h resp=%b", rd1, rp1);

    $display("---- DONE ----");
    repeat (10) @(posedge clk);
    $finish;
  end
